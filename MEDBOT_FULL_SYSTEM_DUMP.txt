_orig_getaddrinfo = socket.getaddrinfo
def getaddrinfo_ipv4(*args, **kwargs):
    responses = _orig_getaddrinfo(*args, **kwargs)
    return [r for r in responses if r[0] == socket.AF_INET]

async def call_gemini_api(prompt: str, context_catalog: str) -> str:
    """استدعاء Gemini مع إعادة محاولة ذكية وتبديل تلقائي عند ضغط الخوادم 503"""
    if not GEMINI_API_KEY:
        return "⚠️ خدمة الذكاء الاصطناعي تحتاج إلى تعيين GEMINI_API_KEY في ملف .env."

    full_prompt = (
        "أنت 'المساعد الطبي الذكي لمنصة MEDBOT' لطلاب كلية الطب البشري.\n"
        "مهامك:\n"
        "1. إرشاد الطلاب للمسار الدقيق لأي ملف بناءً على كشاف المنصة أدناه.\n"
        "2. الإجابة على الاستفسارات الطبية باقتضاب ودقة، مع توضيح المصطلحات الطبية بالإنجليزية والعربية (What, Why, Mechanism).\n\n"
        f"📋 [كشاف المحتويات المتاحة حالياً]:\n{context_catalog}\n\n"
        f"❓ سؤال الطالب:\n{prompt}"
    )

    payload = {"contents": [{"parts": [{"text": full_prompt}]}]}
    candidate_models = ["gemini-flash-lite-latest", "gemma-4-26b-a4b-it"]
    base_host = "https://generativelanguage.googleapis.com"

    async with httpx.AsyncClient(timeout=45.0) as client:
        for model in candidate_models:
            url = f"{base_host}/v1beta/models/{model}:generateContent?key={GEMINI_API_KEY}"
            for attempt in range(2):
                try:
                    response = await client.post(url, json=payload)
                    if response.status_code == 200:
                        data = response.json()
                        parts = data.get("candidates", [])[0].get("content", {}).get("parts", [])
                        texts = [p.get("text", "") for p in parts if "text" in p]
                        if texts:
                            return texts[-1].strip()
                    elif response.status_code in [503, 429]:
                        logger.warning(f"Model {model} returned {response.status_code} (attempt {attempt+1}). Retrying in 1.5s...")
                        await asyncio.sleep(1.5)
                        continue
                    else:
                        logger.error(f"Model {model} returned HTTP {response.status_code}")
                        break
                except Exception as e:
                    logger.warning(f"Network glitch with {model}: {e}")
                    await asyncio.sleep(1.0)
                    continue

    return "⚠️ خوادم المعالجة تشهد ضغطاً مؤقتاً حالياً، يرجى إعادة إرسال السؤال بعد قليل."

async def handle_text(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    uid = update.effective_user.id
    action = context.user_data.get('action')
    text = update.message.text.strip()

    if action == 'await_search_query':
        context.user_data['action'] = None

        # Search Engine v1.1:
        # - searches folders + content
        # - exact/prefix/partial ranking
        # - Arabic normalization
        # - distinguishes FOLDER / EMPTY_FOLDER / CONTENT
        # - returns full paths and real folder content counts
        results = await search_engine.search_library(text, limit=15)

        kb = []

        if results:
            lines = [f"🔍 نتائج البحث عن: **{text}**", ""]

            for result in results:
                result_type = result["type"]
                name = result["name"]
                path = result["path"]
                content_count = result["content_count"]

                if result_type == "CONTENT":
                    cid = result["content_id"]

                    icon = "📄"
                    file_type = result.get("file_type", "")
                    if file_type == "audio":
                        icon = "🎵"
                    elif file_type == "video":
                        icon = "🎬"
                    elif file_type == "photo":
                        icon = "🖼️"

                    kb.append([
                        InlineKeyboardButton(
                            f"{icon} {name}",
                            callback_data=f"dl_{cid}"
                        )
                    ])

                    lines.append(
                        f"📄 **{name}**\n"
                        f"📍 {path}"
                    )

                elif result_type == "EMPTY_FOLDER":
                    fid = result["folder_id"]

                    kb.append([
                        InlineKeyboardButton(
                            f"📭 {name} — فارغ",
                            callback_data=f"f_{fid}"
                        )
                    ])

                    lines.append(
                        f"📭 **{name}** — لا توجد ملفات حاليًا\n"
                        f"📍 {path}"
                    )

                elif result_type == "FOLDER":
                    fid = result["folder_id"]

                    kb.append([
                        InlineKeyboardButton(
                            f"📁 {name}",
                            callback_data=f"f_{fid}"
                        )
                    ])

                    lines.append(
                        f"📁 **{name}** — {content_count} ملف\n"
                        f"📍 {path}"
                    )

            msg_text = "\n\n".join(lines)

        else:
            msg_text = (
                f"📭 لم يتم العثور على نتيجة داخل مكتبة MEDBOT تطابق: **{text}**\n\n"
                "يمكنك تجربة اسم مادة أو محاضرة أو قسم آخر."
            )

        kb.append([
            InlineKeyboardButton("🔍 بحث آخر", callback_data="btn_search")
        ])
        kb.append([
            InlineKeyboardButton("🏠 العودة للرئيسية", callback_data="f_0")
        ])

        await update.message.reply_text(
            msg_text,
            reply_markup=InlineKeyboardMarkup(kb)
        )
        return

    if action == 'await_ai_question':
        context.user_data['action'] = None
        wait_msg = await update.message.reply_text("⏳ جاري البحث والمعالجة الطبية بواسطة الذكاء الاصطناعي...")

        catalog = await db.get_catalog_for_ai()
        ai_response = await call_gemini_api(text, catalog)

        kb = [
            [InlineKeyboardButton("🤖 سؤال جديد للذكاء الاصطناعي", callback_data="btn_ai")],
            [InlineKeyboardButton("🏠 العودة للرئيسية", callback_data="f_0")]
        ]
        await wait_msg.edit_text(ai_response, reply_markup=InlineKeyboardMarkup(kb), parse_mode="Markdown")
        return

    if uid == ADMIN_ID and action == 'input_node_name':
        context.user_data['temp_node_name'] = text
        context.user_data['action'] = None

        kb = [
            [InlineKeyboardButton("📁 مجلد / تفرع عام (سنة، ترم، بلوك، مادة)", callback_data="settype_general")],
            [InlineKeyboardButton("📚 قسم كتب وملازم (PDF)", callback_data="settype_book")],
            [InlineKeyboardButton("🎙️ قسم تسجيلات ومحاضرات صوتية", callback_data="settype_audio")],
            [InlineKeyboardButton("🎬 قسم فيديوهات وشروحات", callback_data="settype_video")],
            [InlineKeyboardButton("❓ بنك أسئلة واختبارات (MCQ)", callback_data="settype_mcq")],
            [InlineKeyboardButton("📑 تفريغات وملخصات", callback_data="settype_summary")]
        ]
        await update.message.reply_text(f"الاسم: **{text}**\nاختر طبيعة هذا التفرع:", reply_markup=InlineKeyboardMarkup(kb), parse_mode="Markdown")

    elif uid == ADMIN_ID and action == 'input_rename':
        fid = context.user_data.get('target_fid')
        pid = await db.get_parent_id(fid)
        await db.update_folder_name(fid, text)
        context.user_data.clear()

        class Dummy:
            def __init__(self, m): self.message = m
            async def edit_message_text(self, *a, **k): await self.message.reply_text(*a, **k)

        await update.message.reply_text(f"✅ تم تعديل الاسم إلى: {text}")
        await render_view(Dummy(update.message), context, pid, uid)

async def handle_media(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    uid = update.effective_user.id
    user = update.effective_user
    msg = update.message

    f_type = "doc"
    if msg.document: f_id, title = msg.document.file_id, msg.document.file_name or "مستند"
    elif msg.audio: f_id, title, f_type = msg.audio.file_id, msg.audio.file_name or msg.audio.title or "تسجيل صوتي", "audio"
    elif msg.voice: f_id, title, f_type = msg.voice.file_id, "رسالة صوتية (Voice)", "audio"
    elif msg.video: f_id, title, f_type = msg.video.file_id, msg.video.file_name or "فيديو", "video"
    elif msg.photo: f_id, title, f_type = msg.photo[-1].file_id, "صورة توضيحية", "photo"
    else: return

    action = context.user_data.get('action')

    if action == 'await_contrib_media':
        fid = context.user_data.get('contrib_fid')
        context.user_data.clear()

        cid = await db.add_contribution(uid, user.full_name, fid, title, f_id, f_type)
        path = await db.get_breadcrumbs(fid)

        kb_student = [[InlineKeyboardButton("🔙 العودة للقسم", callback_data=f"f_{fid}")]]
        await msg.reply_text(
            f"✅ **تم استلام مساهمتك بنجاح يا دكتور!**\n\n📄 الملف: **{title}**\n📍 القسم: **{path}**\n\nتم توجيه الملف للإدارة للاعتماد 🩺",
            reply_markup=InlineKeyboardMarkup(kb_student),
            parse_mode="Markdown"
        )

        if ADMIN_ID != 0:
            caption_admin = (
                f"📩 **مساهمة طلابية جديدة تنتظر الاعتماد**\n"
                f"━━━━━━━━━━━━━━━━━━━━\n"
                f"👤 **الطالب:** {user.full_name} (`{uid}`)\n"
                f"📍 **المسار:** {path}\n"
                f"📄 **العنوان:** {title}"
            )
            kb_admin = [
                [
                    InlineKeyboardButton("✅ اعتماد ونشر فوراً", callback_data=f"adm_appr_{cid}"),
                    InlineKeyboardButton("❌ رفض المساهمة", callback_data=f"adm_rejc_{cid}")
                ]
            ]
            markup = InlineKeyboardMarkup(kb_admin)
            try:
                if f_type == "audio":
                    await context.bot.send_audio(chat_id=ADMIN_ID, audio=f_id, caption=caption_admin, reply_markup=markup, parse_mode="Markdown")
                elif f_type == "video":
                    await context.bot.send_video(chat_id=ADMIN_ID, video=f_id, caption=caption_admin, reply_markup=markup, parse_mode="Markdown")
                elif f_type == "photo":
                    await context.bot.send_photo(chat_id=ADMIN_ID, photo=f_id, caption=caption_admin, reply_markup=markup, parse_mode="Markdown")
                else:
                    await context.bot.send_document(chat_id=ADMIN_ID, document=f_id, caption=caption_admin, reply_markup=markup, parse_mode="Markdown")

                logger.error(f"Failed to forward to admin: {e}")

    if uid == ADMIN_ID:

async def is_descendant(db, ancestor_id: int, folder_id: int) -> bool:
    """Check if folder_id is a descendant of ancestor_id (for cycle prevention)."""
    current = folder_id
    visited = set()
    while current is not None:
        if current == ancestor_id:
            return True
        if current in visited:
            return True
        visited.add(current)
        async with db.execute("SELECT parent_id FROM folders WHERE id = ?", (current,)) as cur:
            row = await cur.fetchone()
        if not row:
            break
        current = row[0]
    return False

        new_pid = None if new_parent_id == 0 else new_parent_id

        )

        await db.execute("""
            UPDATE contributions
            SET status = 'approved'
            WHERE id = ? AND status = 'pending'
        """, (contrib_id,))
        await db.commit()
        return (fid, title, file_id, f_type, uid)
    except Exception:
        await db.rollback()

async def ai_registry_add(provider: str, model: str = None, endpoint: str = None, availability: str = None, auth_status: str = None, latency_ms: float = None, success_rate: float = None, capabilities: str = None, notes: str = None):
    """Idempotently register an AI provider/model/endpoint.

    Canonical identity:
        (provider, model, endpoint)

    Incoming NULL values never erase existing non-NULL values.
    """
    db = await get_db()
    await db.execute("""
        INSERT INTO ai_registry (
            provider, model, endpoint, availability, auth_status,
            latency_ms, success_rate, capabilities, last_test, notes
        )
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, CURRENT_TIMESTAMP, ?)
        ON CONFLICT(provider, model, endpoint) DO UPDATE SET
            availability = COALESCE(excluded.availability, ai_registry.availability),
            auth_status = COALESCE(excluded.auth_status, ai_registry.auth_status),
            latency_ms = COALESCE(excluded.latency_ms, ai_registry.latency_ms),
            success_rate = COALESCE(excluded.success_rate, ai_registry.success_rate),
            capabilities = COALESCE(excluded.capabilities, ai_registry.capabilities),
            notes = COALESCE(excluded.notes, ai_registry.notes),
            last_test = CURRENT_TIMESTAMP
    """, (
        provider, model, endpoint, availability, auth_status,
        latency_ms, success_rate, capabilities, notes
    ))
    await db.commit()
    await db.close()

async def ai_registry_mark_success(
    id: int,
    latency_ms: float = None,
    notes: str = "verified",
):
    """Record a successful provider/model operation.

    State semantics:
    - availability -> VERIFIED
    - auth_status -> valid
    - last_success -> current UTC timestamp
    - last_test -> current UTC timestamp
    - error_category -> NULL
    """
    db = await get_db()
    await db.execute("""
        UPDATE ai_registry
        SET
            availability = 'VERIFIED',
            auth_status = 'valid',
            latency_ms = COALESCE(?, latency_ms),
            last_success = CURRENT_TIMESTAMP,
            last_test = CURRENT_TIMESTAMP,
            error_category = NULL,
            notes = ?
        WHERE id = ?
    """, (latency_ms, notes, id))

    await db.close()


async def ai_registry_mark_failure(
    id: int,
    availability: str,
    error_category: str,
    auth_status: str = None,
    notes: str = None,
):
    """Record a failed provider/model operation.

    The caller supplies the already-classified availability state.
    This function never invents or reclassifies errors.
    """
    allowed_states = {
        "RATE_LIMITED",
        "TIMEOUT",
        "AUTH_FAILED",
        "UPSTREAM_ERROR",
        "UNAVAILABLE",
    }

    if availability not in allowed_states:
        raise ValueError(
            f"Invalid failure availability state: {availability!r}"
        )

    db = await get_db()
    await db.execute("""
        UPDATE ai_registry
        SET
            availability = ?,
            auth_status = COALESCE(?, auth_status),
            last_failure = CURRENT_TIMESTAMP,
            last_test = CURRENT_TIMESTAMP,
            error_category = ?,
            notes = COALESCE(?, notes)
        WHERE id = ?
    """, (
        availability,
        auth_status,
        error_category,
        notes,
        id,
    ))
    await db.commit()
    await db.close()

                    result["auth_status"] = "unknown"

                    result["error"] = f"HTTP {response.status_code}"

                raise Exception(f"Gemini error: HTTP {response.status_code}")

class AIFactory:
    """
    Central provider catalog and factory.

    The catalog is intentionally limited to provider implementations
    that already exist in this module. Stage 2 does not introduce
    speculative providers or models.
    """

    PROVIDER_CATALOG = {
        "google_gemini": {
            "env_var": "GEMINI_API_KEY",
            "provider_class": GeminiProvider,
            "models": (
                "gemini-flash-lite-latest",
                "gemma-4-26b-a4b-it",
            ),
        },
        "openrouter": {
            "env_var": "OPENROUTER_API_KEY",
            "provider_class": OpenRouterProvider,
            "models": (
                "deepseek/deepseek-v4-flash",
                "nvidia/llama-3.1-nemotron-ultra",
                "nexusflow/nex-n2.5-mini",
            ),
        },
        "openai": {
            "env_var": "OPENAI_API_KEY",
            "provider_class": OpenAIProvider,
            "models": (
                "gpt-4o-mini",
            ),
        },
    }

    @classmethod
    def create_providers(
        cls,
        gemini_api_key: str = "",
        openrouter_api_key: str = "",
        openai_api_key: str = "",
    ) -> list[AIProvider]:
        """
        Create providers from the centralized catalog.

        Existing public arguments are preserved for backward compatibility.
        No API key values are logged or returned by this method.
        """
        keys = {
            "GEMINI_API_KEY": (gemini_api_key or "").strip(),
            "OPENROUTER_API_KEY": (openrouter_api_key or "").strip(),
            "OPENAI_API_KEY": (openai_api_key or "").strip(),
        }

        providers = []

        for spec in cls.PROVIDER_CATALOG.values():
            api_key = keys.get(spec["env_var"], "")
            if not api_key:
                continue

            provider_class = spec["provider_class"]

            for model in spec["models"]:
                if provider_class is GeminiProvider:
                    providers.append(provider_class(api_key, model))
                elif provider_class is OpenRouterProvider:
                    providers.append(provider_class(api_key, model))
                elif provider_class is OpenAIProvider:
                    providers.append(provider_class(api_key, model))
                else:
                    raise RuntimeError(
                        f"Unsupported provider implementation: {provider_class!r}"
                    )

        return providers

    @classmethod
    def discover_provider_env_status(cls) -> dict:
        """
        Safely report which supported provider environment variables exist.

        Only boolean presence is returned. Secret values are never returned.
        """
        import os

        return {
            provider_name: bool(os.getenv(spec["env_var"], "").strip())
            for provider_name, spec in cls.PROVIDER_CATALOG.items()
        }

    @classmethod
    def discover_providers_from_env(cls) -> dict:
        """
        Backward-compatible environment discovery.

        Returns only supported environment variables that are present.
        This preserves the previous public behavior while keeping the
        supported provider catalog centralized.
        """
        import os

        env_keys = {}

        for spec in cls.PROVIDER_CATALOG.values():
            var = spec["env_var"]
            val = os.getenv(var, "").strip()
            if val:
                env_keys[var] = val

        return env_keys

def is_transient_error(e: Exception) -> bool:
    msg = str(e).lower()
    if "timeout" in msg:
        return True
    if "429" in msg or "rate limit" in msg:
        return True
    if "500" in msg or "502" in msg or "503" in msg:
        return True
    if "connection" in msg and ("refused" in msg or "reset" in msg):
        return True
    return False

def is_permanent_error(e: Exception) -> bool:
    msg = str(e).lower()
    if "401" in msg or "403" in msg:
        return True
    if "invalid" in msg and ("key" in msg or "api" in msg):
        return True
    if "not found" in msg or "model not found" in msg or "model_not_found" in msg:
        return True
    return False

class AIRouter:
    def __init__(self, gemini_api_key: str = "", openrouter_api_key: str = "", openai_api_key: str = "", providers: list = None):
        if providers is not None:
            self.providers = providers
        else:
            self.providers = AIFactory.create_providers(gemini_api_key, openrouter_api_key, openai_api_key)
        self.validator = GroundingValidator()
        self._registry_cache = {}

    async def discover_and_register(self):
        logger.info("Starting AI provider discovery...")
        tested = []
        for provider in self.providers:
            result = await provider.test_connectivity()
            tested.append(result)
            logger.info(f"Provider {provider.name}/{provider.model}: {result['availability']} ({result['latency_ms']}ms)")
            await db.ai_registry_add(
                provider=provider.name,
                model=provider.model,
                endpoint=provider.endpoint,
                availability=result["availability"],
                auth_status=result.get("auth_status"),
                latency_ms=result.get("latency_ms"),
                capabilities="text_generation",
                notes=result.get("error")
            )
        available = [r for r in tested if r["availability"] in (PROVIDER_STATUS_AVAILABLE, PROVIDER_STATUS_VERIFIED)]
        available.sort(key=lambda r: r.get("latency_ms", 9999))
        for i, r in enumerate(available):
            logger.info(f"Available provider #{i+1}: {r['provider']}/{r['model']} - {r['latency_ms']}ms")
        return tested

    async def _get_healthy_providers_ranked(self) -> list[AIProvider]:
        try:
            registry_rows = await db.ai_registry_get_healthy()
            healthy_endpoints = {}
            for row in registry_rows:
                rid, provider_name, model, endpoint = row[:4]
                availability = row[4]
                latency = row[6] or 9999
                success_rate = row[7] or 0.0
                healthy_endpoints[f"{provider_name}|{model}"] = {
                    "rank": success_rate / max(latency, 1),
                    "rid": rid,
                    "availability": availability,
                    "latency": latency
                }
            scored = []
            for p in self.providers:
                key = f"{p.name}|{p.model}"
                if key in healthy_endpoints:
                    scored.append((healthy_endpoints[key]["rank"], p))
                else:
                    scored.append((0.5, p))
            scored.sort(key=lambda x: x[0], reverse=True)
            return [p for _, p in scored]
        except Exception:
            return self.providers

    async def generate_with_failover(self, prompt: str, system_prompt: str = None) -> tuple[Optional[str], Optional[str], float, Optional[str]]:
        ranked = await self._get_healthy_providers_ranked()
        last_error = None
        for attempt, provider in enumerate(ranked):
            try:
                start = time.time()
                response = await provider.generate(prompt, system_prompt)
                elapsed = (time.time() - start) * 1000
                logger.info(f"AI response from {provider.name}/{provider.model}: {round(elapsed, 1)}ms (attempt {attempt+1})")
                try:
                    rid = await self._get_registry_id(provider)
                    if rid:
                        await db.ai_registry_mark_success(
                            rid,
                            latency_ms=round(elapsed, 1),
                            notes="verified"
                        )
                except Exception:
                    pass
                return response, provider.name, round(elapsed, 1), None
            except Exception as e:
                last_error = e
                if is_permanent_error(e):
                    logger.warning(f"Permanent error from {provider.name}/{provider.model}: {e}")
                    try:
                        rid = await self._get_registry_id(provider)
                        if rid:
                            await db.ai_registry_mark_failure(
                            rid,
                            availability=PROVIDER_STATUS_AUTH_FAILED,
                            error_category="auth_failed",
                            auth_status="invalid",
                            notes="authentication failure"
                        )
                    except Exception:
                        pass
                    continue
                if is_transient_error(e) and attempt < len(ranked) - 1:
                    logger.warning(f"Transient error from {provider.name}/{provider.model}: {e}. Trying next provider...")
                    try:
                        rid = await self._get_registry_id(provider)
                        if rid:
                            ec = "rate_limit" if "429" in str(e) else "timeout" if "timeout" in str(e).lower() else "upstream_error"
                            failure_state = (
                            PROVIDER_STATUS_RATE_LIMITED
                            if ec == "rate_limit"
                            else PROVIDER_STATUS_TIMEOUT
                            if ec == "timeout"
                            else PROVIDER_STATUS_UPSTREAM_ERROR
                        )

                        await db.ai_registry_mark_failure(
                            rid,
                            availability=failure_state,
                            error_category=ec,
                            notes=f"transient failure: {ec}"
                        )
                    except Exception:
                        pass
                    await asyncio.sleep(1.0)
                    continue
                logger.warning(f"Provider {provider.name}/{provider.model} failed: {e}")
                try:
                    rid = await self._get_registry_id(provider)
                    if rid:
                        await db.ai_registry_mark_failure(
                            rid,
                            availability=PROVIDER_STATUS_UPSTREAM_ERROR,
                            error_category="upstream_error",
                            notes="provider failure"
                        )
                except Exception:
                    pass
        return None, None, 0, f"All providers failed. Last error: {last_error}"

    async def _get_registry_id(self, provider) -> Optional[int]:
        cache_key = f"{provider.name}|{provider.model}"
        if cache_key in self._registry_cache:
            return self._registry_cache[cache_key]
        try:
            rows = await db.ai_registry_get_all()
            for row in rows:
                if row[1] == provider.name and row[2] == provider.model:
                    self._registry_cache[cache_key] = row[0]
                    return row[0]
        except Exception:
            pass
        return None

import re
import unicodedata
from typing import Any

import aiosqlite

DB_NAME = "medbot_v2.sqlite3"


def normalize_text(value: str) -> str:
    """
    Normalize Arabic/English search text without destroying the original
    database values.

    Arabic:
      - remove tashkeel
      - normalize alef variants
      - normalize ya/alef maqsura
      - normalize ta marbuta
    English:
      - lowercase
      - normalize Unicode
      - collapse whitespace
    """
    if not value:
        return ""

    text = unicodedata.normalize("NFKC", str(value)).strip().lower()

    # Remove Arabic tashkeel.
    text = re.sub(r"[\u0610-\u061A\u064B-\u065F\u0670\u06D6-\u06ED]", "", text)

    # Arabic normalization.
    text = text.translate(str.maketrans({
        "أ": "ا",
        "إ": "ا",
        "آ": "ا",
        "ٱ": "ا",
        "ى": "ي",
        "ة": "ه",
    }))

    # Collapse whitespace.
    text = re.sub(r"\s+", " ", text)

    return text


def _like_pattern(value: str) -> str:
    """
    Escape LIKE wildcards so user input is treated as text.
    """
    value = value.replace("\\", "\\\\")
    value = value.replace("%", "\\%")
    value = value.replace("_", "\\_")
    return value


async def _build_folder_path(db: aiosqlite.Connection, folder_id: int) -> str:
    """
    Build a human-readable folder path.

    Root is represented internally by NULL parent_id.
    Cycle protection is included so corrupted data cannot loop forever.
    """
    parts = []
    current = folder_id
    visited = set()

    while current is not None:
        if current in visited:
            parts.append("[CYCLE]")
            break

        visited.add(current)

        async with db.execute(
            "SELECT parent_id, name FROM folders WHERE id = ?",
            (current,),
        ) as cur:
            row = await cur.fetchone()

        if not row:
            break

        parent_id, name = row
        parts.append(name)
        current = parent_id

    parts.reverse()
    return "الرئيسية 🏠" + ((" ⬅️ " + " ⬅️ ".join(parts)) if parts else "")


async def _folder_content_count(
    db: aiosqlite.Connection,
    folder_id: int,
) -> int:
    async with db.execute(
        "SELECT COUNT(*) FROM content WHERE folder_id = ?",
        (folder_id,),
    ) as cur:
        row = await cur.fetchone()

    return int(row[0] if row else 0)


async def search_library(
    keyword: str,
    limit: int = 15,
) -> list[dict[str, Any]]:
    """
    Search Engine v1.

    Search scope:
      1. Folder names
      2. Content titles

    Ranking:
      0 = exact
      1 = prefix
      2 = partial

    Result types:
      FOLDER
      CONTENT
      EMPTY_FOLDER

    The database remains the source of truth.
    """

    query = normalize_text(keyword)

    if not query:
        return []

    limit = max(1, min(int(limit), 50))
    escaped = _like_pattern(query)

    db = await aiosqlite.connect(DB_NAME)

    try:
        # Fetch all candidate folders/content records. The dataset is
        # intentionally small at this stage, and normalization is performed
        # in Python so Arabic matching is reliable without changing stored data.
        async with db.execute(
            """
            SELECT
                f.id,
                f.parent_id,
                f.name,
                f.node_type,
                COUNT(c.id) AS content_count
            FROM folders f
            LEFT JOIN content c ON c.folder_id = f.id
            GROUP BY f.id
            ORDER BY f.id ASC
            """
        ) as cur:
            folders = await cur.fetchall()

        async with db.execute(
            """
            SELECT
                c.id,
                c.folder_id,
                c.title,
                c.file_type,
                (
                    SELECT COUNT(*)
                    FROM content fc
                    WHERE fc.folder_id = c.folder_id
                ) AS folder_content_count
            FROM content c
            ORDER BY c.id DESC
            """
        ) as cur:
            contents = await cur.fetchall()

        results: list[dict[str, Any]] = []

        # Folder results.
        for folder_id, parent_id, name, node_type, content_count in folders:
            normalized_name = normalize_text(name)

            if normalized_name == query:
                rank = 0
            elif normalized_name.startswith(query):
                rank = 1
            elif query in normalized_name:
                rank = 2
            else:
                continue

            path = await _build_folder_path(db, folder_id)

            result_type = "FOLDER"
            if int(content_count) == 0:
                result_type = "EMPTY_FOLDER"

            results.append({
                "type": result_type,
                "result_type": result_type,
                "id": folder_id,
                "folder_id": folder_id,
                "title": name,
                "name": name,
                "path": path,
                "content_count": int(content_count),
                "node_type": node_type,
                "rank": rank,
                "match_field": "folder_name",
            })

        # Content results.
        for content_id, folder_id, title, file_type, folder_content_count in contents:
            normalized_title = normalize_text(title)

            if normalized_title == query:
                rank = 0
            elif normalized_title.startswith(query):
                rank = 1
            elif query in normalized_title:
                rank = 2
            else:
                continue

            path = await _build_folder_path(db, folder_id)

            results.append({
                "type": "CONTENT",
                "result_type": "CONTENT",
                "id": content_id,
                "content_id": content_id,
                "folder_id": folder_id,
                "title": title,
                "name": title,
                "path": path,
                "content_count": int(folder_content_count),
                "file_type": file_type,
                "rank": rank,
                "match_field": "content_title",
            })

        # Deterministic ordering:
        # exact → prefix → partial → folders before content → title → id.
        results.sort(
            key=lambda item: (
                item["rank"],
                0 if item["result_type"] in ("FOLDER", "EMPTY_FOLDER") else 1,
                normalize_text(item["title"]),
                item["id"],
            )
        )

        return results[:limit]

    finally:
        await db.close()


async def search_library_summary(keyword: str, limit: int = 15) -> dict[str, Any]:
    """
    Stable API for Telegram/AI layers.
    """
    results = await search_library(keyword, limit)

    return {
        "query": keyword,
        "normalized_query": normalize_text(keyword),
        "result_count": len(results),
        "results": results,
    }
